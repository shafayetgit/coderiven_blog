Scroll to Specific Section in a React App

If scrolling to a specific section via a hash doesn't always work, it could be due to how your React application handles routing or rendering. Here's a reliable solution using useLocation and useEffect.


// import { useLocation } from "react-router"; // do not forget to import it from react-router (v7) or react-router-dom

  const location = useLocation();

  useEffect(() => {
    if (location.hash) {
      const scrollToElement = (hash, retries = 10) => {
        const element = document.querySelector(hash);
        if (element) {
          element.scrollIntoView({ behavior: "smooth" });
        } else if (retries > 0) {
          setTimeout(() => scrollToElement(hash, retries - 1), 100);
        }
      };

      scrollToElement(location.hash);
    }
  }, [location.hash]);

          scrollToElement(location.hash);

    };

    export default useScrollToHashSection;
